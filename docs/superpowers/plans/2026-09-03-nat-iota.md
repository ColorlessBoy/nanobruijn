```python
def reduce_rec(self, const_name: NamePtr, const_levels: LevelsPtr, args: list) -> ExprPtr | None:
    """iota：按 major 前提的构造子分派 rec 规则（镜像 Rust reduce_rec）。

    仅对带规则的 RecursorDecl 生效——axiom 形状的 rec（如 fol 的 Or.rec）
    永不归约。v1 不含 K-target / NatLit / eta-struct 分支。
    """
    rec = self.env.get_recursor(const_name)
    if rec is None or not rec.rules:
        return None
    if len(args) <= rec.major_idx():
        return None
    major = self.whnf(args[rec.major_idx()])
    major_ctor, major_args = self.ctx.unfold_apps(major)
    rule = next((r for r in rec.rules if r.ctor_name == major_ctor), None)
    if rule is None:
        return None
    extra = len(major_args) - rule.ctor_telescope_size_wo_params
    prefix = rec.num_params + rec.num_motives + rec.num_minors
    r = self.ctx.subst_expr_levels(rule.val, rec.info.uparams, const_levels)
    r = self.ctx.foldl_apps(r, args[:prefix])
    r = self.ctx.foldl_apps(r, major_args[extra:])
    return self.ctx.foldl_apps(r, args[rec.major_idx() + 1:])
```

```python
def _build_inductive_block(core, lines):
    """装配 inductive/ctor/rec 块并交内核检查。

    v1 限制：无参数、无索引、Type 排序、字段类型不含括号。
    """
    from ..dag import TcCtx  # noqa: F401（core.ctx 已是 TcCtx）
    from ..env import (ConstructorData, ConstructorDecl, InductiveData,
                       InductiveDecl, RecursorData, RecursorDecl, RecRule)
    from ..inductive import check_inductive_declaration, core_has_const
    from types import SimpleNamespace

    m = IND_RE.match(lines[0])
    if m is None:
        raise ValueError(f"fol: inductive 块首行无法解析: {lines[0]!r}")
    ind_name_str, ind_ty = m.group(1), m.group(2)
    ctor_lines = [l for l in lines[1:] if l.startswith('ctor ')]
    rec_lines = [l for l in lines[1:] if l.startswith('rec ')]
    if len(rec_lines) > 1:
        raise ValueError("fol: 一个 inductive 块最多一条 rec")
    ctor_names = [CTOR_RE.match(l).group(1) for l in ctor_lines]

    ind_ty_e = parse_expr(core, ind_ty)
    ind_ptr = core.name_to_ptr(ind_name_str)
    up_empty = core.dag.insert_uparams(())
    ind_info = DeclarInfo(name=ind_ptr, uparams=up_empty, ty=ind_ty_e.core)
    ind_data = InductiveData(info=ind_info, all_ctor_names=tuple(core.name_to_ptr(c) for c in ctor_names),
                             all_inductive_infos=(ind_ptr,), num_params=0, num_indices=0,
                             num_nested=0, is_rec=False, is_reflexive=False)
    ind_decl = InductiveDecl(info=ind_info, inductives=(ind_data,),
                             constructors=(), recursors=())
    # 先入 env：ctor/rec 类型里要引用归纳常量
    core.env.declars[ind_ptr] = ind_decl

    ctor_datas = []
    for l in ctor_lines:
        cm = CTOR_RE.match(l)
        if cm is None:
            raise ValueError(f"fol: ctor 行无法解析: {l!r}")
        c_name, fields_s, c_ty = cm.group(1), cm.group(2) or "", cm.group(3)
        c_ptr = core.name_to_ptr(c_name)
        ty_e = parse_expr(core, c_ty)
        for fname, fty in reversed(_split_fields(fields_s)):
            ty_e = core.ctx.mk_pi(core.name_to_ptr(fname),
                                  __import__('py_nanobruijn.binder_style', fromlist=['BinderStyle']).BinderStyle.DEFAULT,
                                  parse_expr(core, fty), ty_e)
        c_info = DeclarInfo(name=c_ptr, uparams=up_empty, ty=ty_e.core)
        n_fields = len(_split_fields(fields_s))
        ctor_datas.append(ConstructorData(info=c_info, cidx=len(ctor_datas),
                                          num_params=0, num_fields=n_fields,
                                          inductive_name=ind_ptr, inductive_names=(ind_ptr,)))
        core.env.declars[c_ptr] = ConstructorDecl(info=c_info, data=ctor_datas[-1])

    rec_data = None
    if rec_lines:
        rm = REC_RE.match(rec_lines[0])
        if rm is None:
            raise ValueError(f"fol: rec 行无法解析: {rec_lines[0]!r}")
        rec_name, rec_uparams, rec_ty = rm.group(1), rm.group(2) or "", rm.group(3)
        rec_up = tuple(p.strip() for p in rec_uparams.split(',')) if rec_uparams else ()
        rec_ptr = core.name_to_ptr(rec_name)
        rec_ty_e = parse_expr(core, rec_ty)
        rules = _synthesize_rules(core, ind_ptr, rec_ptr, rec_up, ctor_datas)
        rec_info = DeclarInfo(name=rec_ptr,
                              uparams=core.dag.insert_uparams(
                                  tuple(core.dag.insert_level(Level.param(core.name_to_ptr(p)))
                                        for p in rec_up)),
                              ty=rec_ty_e.core)
        rec_data = RecursorData(info=rec_info, num_params=0, num_indices=0,
                                num_motives=1, num_minors=len(ctor_datas),
                                rules=tuple(rules), all_inductives=(ind_ptr,), k=False)
        core.env.declars[rec_ptr] = RecursorDecl(info=rec_info, data=rec_data)

    ind_decl = InductiveDecl(info=ind_info, inductives=(ind_data,),
                             constructors=tuple(ctor_datas),
                             recursors=(rec_data,) if rec_data else ())
    core.env.declars[ind_ptr] = ind_decl
    export = SimpleNamespace(dag=core.dag)
    check_inductive_declaration(export, ind_decl, core.env.declars)
```

```python
def _split_fields(s):
    """"(n : Nat) (m : Nat)" → [("n", "Nat"), ("m", "Nat")]（v1：字段类型不含括号）"""
    out = []
    for item in FIELD_RE.findall(s or ""):
        name, _, ty = item.partition(':')
        out.append((name.strip(), ty.strip()))
    return out


def _synthesize_rules(core, ind_ptr, rec_ptr, rec_uparams, ctor_datas):
    """受限形状的 rec 规则综合。

    规则 rhs 上下文 = [motive, minor_1..minor_k, f_1..f_n]：
    motive=Var(k)、minor_j=Var(k-j)、字段 f_j=Var(n-j)。
    递归字段（类型提到归纳常量）在字段参数后追加递归调用参数。
    """
    from ..inductive import core_has_const
    k = len(ctor_datas)
    rec_c = core.ctx.mk_const(rec_ptr,
                              core.dag.insert_uparams(
                                  tuple(core.dag.insert_level(Level.param(core.name_to_ptr(p)))
                                        for p in rec_uparams)))
    rules = []
    for i, ctor in enumerate(ctor_datas):
        # 剥 ctor 类型的字段 telescope（无参数：全部 Pi binder 都是字段）
        field_types = []
        cur = ExprPtr.closed(ctor.info.ty)
        while True:
            v = core.ctx.view_expr(cur)
            if v.tag != 'Pi':
                break
            field_types.append(v.children[2])
            cur = v.children[3]
        n = len(field_types)
        minor_var = core.ctx.mk_var(k - 1 - i)  # minor_i（i 从 0 计）
        args = []
        for j, fty in enumerate(field_types):
            f_var = core.ctx.mk_var(n - 1 - j)
            args.append(f_var)
            if core_has_const(core.dag, fty.core, {ind_ptr}):
                rec_app = core.ctx.mk_app(rec_c, core.ctx.mk_var(k))
                for j2 in range(k):
                    rec_app = core.ctx.mk_app(rec_app, core.ctx.mk_var(k - 1 - j2))
                args.append(core.ctx.mk_app(rec_app, f_var))
        rhs = core.ctx.foldl_apps(minor_var, args)
        rules.append(RecRule(ctor_name=ctor.info.name,
                             ctor_telescope_size_wo_params=n,
                             val=rhs.core))
    return rules
```

（`minor_var` 的索引：minor_1 = Var(k-1)、minor_i = Var(k-i)，i 从 1 计；
代码中 `k - 1 - i`，i 从 0 计——等价。）

```python
class ReductionStep(NamedTuple):
    before: ExprPtr
    after: ExprPtr
    kind: str  # 'beta' | 'delta' | 'iota'


def _head_rec_args(tc: TypeChecker, e: ExprPtr):
    """头是带规则的 RecursorDecl 时返回 (const_name, levels, args)，否则 None。"""
    fun, args = tc.ctx.unfold_apps(e)
    expr = tc.ctx.dag.get_expr(fun.core)
    if expr.tag != 'Const':
        return None
    rec = tc.env.get_recursor(expr.children[0])
    if rec is None or not rec.rules or len(args) <= rec.major_idx():
        return None
    return expr.children[0], expr.children[1], args


def reduce_steps(tc: TypeChecker, e: ExprPtr) -> list[ReductionStep]:
    steps: list[ReductionStep] = []
    cursor = e
    while True:
        head = _head_rec_args(tc, cursor)
        if head is not None:
            r = tc.reduce_rec(*head)
            if r is not None:
                steps.append(ReductionStep(cursor, r, "iota"))
                cursor = r
                continue
        unfolded = tc.unfold_def(cursor)
        if unfolded is not None:
            steps.append(ReductionStep(cursor, unfolded, "delta"))
            cursor = unfolded
            continue
        r = tc.whnf_no_unfolding(cursor)
        if r != cursor:
            steps.append(ReductionStep(cursor, r, "beta"))
            cursor = r
            continue
        # 头卡住：下探第一个可归约的参数（succ (rec … one) 场景）
        fun, args = tc.ctx.unfold_apps(cursor)
        if args:
            for i, a in enumerate(args):
                sub = reduce_steps(tc, a)
                if sub:
                    new_args = list(args)
                    new_args[i] = sub[-1].after
                    next_cur = tc.ctx.foldl_apps(fun, new_args)
                    steps.extend(ReductionStep(s.before, s.after, s.kind) for s in sub)
                    cursor = next_cur
                    break
            else:
                break
            continue
        break
    return steps
```
